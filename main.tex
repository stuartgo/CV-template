%% Section header (bold caps + full-width rule)
\newcommand{\cvsection}[1]{%
  \vspace{10pt}%
  {\Large\bfseries\MakeUppercase{#1}}\\[-8pt]%
  {\rule{\linewidth}{1.5pt}}%
  \vspace{1pt}%
}

%% Job / degree title
\newcommand{\jobtitle}[1]{{\large\bfseries #1}}

%% Company / institution in blue
\newcommand{\company}[2][]{%
  {\small\bfseries%
  \ifx&#1&%
    \textcolor{accentblue}{#2}%
  \else%
    \href{#1}{\textcolor{accentblue}{#2}}%
  \fi}%
}

%% Grey italic qualifier line under a job title (FTE, contract nature, etc.)
\newcommand{\rolenote}[1]{%
  \vspace{1pt}{\footnotesize\textcolor{gray}{\textit{#1}}}\vspace{1pt}%
}

\definecolor{availbg}{HTML}{EAF6FB}   % pale tint of your accentblue

%% Optional highlighted availability/callout box under the header
\newcommand{\availability}[1]{%
  \par\vspace{5pt}%
  \noindent
  \begin{tabular}{@{}!{\color{accentblue}\vrule width 2.2pt}p{\dimexpr0.52\linewidth \relax}@{}}
    \cellcolor{availbg}\small #1
  \end{tabular}%
  \par\vspace{3pt}%
}

%% Date + place line
\newcommand{\dateplace}[3]{%
  \vspace{1pt}%
  {\footnotesize\textcolor{gray}{%
    \faBuilding\;#3%
    \hspace{0.5em}%
    \faMapMarker\;#2%
    \hspace{0.5em}%
    \faCalendar\;#1%
  }}%
  \vspace{1pt}%
}

%% Language proficiency dots (filled = blue, empty = gray)
%% Usage: \langdots{<filled count>}
\newcommand{\langdots}[1]{%
  \foreach \x in {1,...,5}{%
    \begin{tikzpicture}[baseline=0.55ex]
      \ifnum\x>#1
        \fill[dotgray] (0,0) circle (2pt);
      \else
        \fill[accentblue] (0,0) circle (2pt);
      \fi
    \end{tikzpicture}\hspace{0.5pt}%
  }%
}

%% Skill badge (rounded-rectangle tag)
\newcommand{\skillbadge}[1]{%
  \begin{tikzpicture}[baseline=(t.base)]
    \node[draw=dotgray, rounded corners=2.5pt,
          inner xsep=5pt, inner ysep=3pt,
          font=\small\sffamily](t){#1};
  \end{tikzpicture}\hspace{1pt}%
}

%% Highlighted text block (orange/yellow background)
%% Usage: \highlight{text}
\newcommand{\highlight}[1]{%
  \colorbox{highlightbg}{\parbox{\dimexpr\linewidth-2\fboxsep}{#1}}%
}

%% Clickable underlined project bullet text
\newcommand{\projlink}[2]{\href{#1}{\uline{#2}}}
